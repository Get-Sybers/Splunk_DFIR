      <title>Average Hourly Authentications Per Domain</title>
      <table>
        <title>Displays the average hourly Kerberos authentications received by each domain for a specified period of time (preferably the last 5-7 days). Additionally, this panel calculates the Hourly Threshold for each domain to be used for hunting anomalies.</title>
        <search>
          <query>index=bro sourcetype=bro_kerberos client!="" success=true 
| rex field=client (?&lt;DomainName&gt;"[^\/]([^/]+)$") 
| eval DomainName=upper(DomainName) 
| eval Day=strftime(_time, "%D") 
| eval Hour=strftime(_time, "%H") 
| stats count by Day Hour DomainName
| dedup Day Hour DomainName
| sort Day DomainName Hour
| stats avg(count) as AvgHourlyAuths by DomainName
| eval HourlyThreshold=(AvgHourlyAuths*0.3)
| table DomainName AvgHourlyAuths HourlyThreshold
| outputlookup Base-HourlyThreshold.csv</query>
          <earliest>$time0.earliest$</earliest>
          <latest>$time0.latest$</latest>
        </search>

      <table>
        <title>Displays the usual login hours for each domain in a specified period of time (preferably the last 5-7 days).</title>
        <search>
          <query>index=bro  sourcetype=bro_kerberos client!="" success=true
| rex field=client (?&lt;DomainName&gt;"[^\/]([^/]+)$") 
| eval DomainName=upper(DomainName)
| eval Day=strftime(_time, "%D") 
| eval Hour=strftime(_time, "%H:00") 
| stats count by Day Hour DomainName 
| lookup Base-HourlyThreshold.csv DomainName
| eval IsUsualHour=if(count&gt;HourlyThreshold, 1, 0)
| dedup Hour DomainName IsUsualHour
| sort DomainName Day Hour 
| outputlookup Base-UsualLoginHours.csv
| stats values(Hour) as UsualLoginHours by DomainName</query>
