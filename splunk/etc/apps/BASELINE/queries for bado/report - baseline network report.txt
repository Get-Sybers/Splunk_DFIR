|multisearch [ search index=host  "Command.Module"=GetDomainComputers
             | rex field=Data.DNSName "^(?<Client_Name>[^.]+)"
             | rex field=Data.DNSName "^[^.]+\.(?<DomainName>.+)"
             | eval source = "host"
             | eval Client_IP='Data.IPAddress']
             [ search index=bro sourcetype=bro_dns  query!="*\.arpa" query!="" answers{}!="0.0.0.0" query!="0.0.0.0"
             | rex field=answers{} "(?<Client_IP>\b(?:\d{1,3}\.){3}\d{1,3}\b)"
             | rex field=query "^(?<Client_Name>[^.]+)"
             | rex field=query "^[^.]+\.(?<DomainName>.+)"
             | eval source="bro_dns"]
| eval DomainName=upper(DomainName), Client_IP=upper(Client_IP), Client_Name=upper(Client_Name)
| dedup DomainName, Client_IP, Client_Name, source
| table DomainName, Client_IP, Client_Name, source
| sort Client_IP 
| outputcsv BASELINE_Network_Report.csv
